\node[data] (raw) {%

};

% 2. Processing
\node[data, below=of raw] (proc) {%
  \textbf{Data Processing}\\[2pt]

  \textbf{First innings:} 129,322 snapshots \textbullet{} \textbf{Second innings:} 112,935 snapshots\\
  Right-censoring identified: 49.2\% of 2nd-innings snapshots (winning teams)
};

% 3. Feature engineering
\node[feat, below=of proc] (feat) {%
  \textbf{Feature Engineering — V2 Pipeline (46 features)}\\[2pt]
  \begin{tabular}{@{}ll@{\hspace{8pt}}ll@{}}
    Match state (6) & Elo ratings (5) & Venue history (6) & Momentum (6) \\
    Player stats (9) & DLS-derived (4) & Phase / Context (10) & \\
  \end{tabular}\\[3pt]
  All features computed \emph{strictly chronologically} — no look-ahead\\
  V3 extension: 9 additional H2H, toss-advantage \& context features (55 total)
};

% 4. Split
\node[split, below=of feat] (split) {%
  \textbf{Temporal Three-Way Split} \quad (strictly chronological)\\[2pt]
  \textbf{60\% Train} (pre-2018, $\approx$78k) \quad
  \textbf{20\% Calibration} (2018--2021, $\approx$26k) \quad
  \textbf{20\% Test} (2022--2026, 542 matches)
};

%% ── Two branches after split ──────────────────────────────

% DLS branch (left)
\node[dls, below left=0.8cm and 0.5cm of split] (dls) {%
  \textbf{DLS Baseline}\\[2pt]
  Fit $Z(u,w)$ via\\
  \texttt{scipy.optimize}\\
  on 3,086 matches\\[2pt]
  Resource table $R(u,w)$
};

% ML branch (right)
\node[model, below right=0.8cm and -0.5cm of split] (ml) {%
  \textbf{ML Models}\\[2pt]
  XGBoost \quad LightGBM \quad CatBoost\\[2pt]
  \textbf{Optuna} Bayesian HPO\\
  100 / 50 / 50 trials\\
  Early stopping on calibration set
};

%% ── Rejoin ────────────────────────────────────────────────

% Extended architectures
\node[ext, below=3.2cm of split] (ext) {%
  \textbf{Extended Architectures}\\[2pt]
  \textbf{Stacking Ensemble} — Ridge meta-learner on XGB + LGB + CAT calibration predictions ($\alpha=1000$)\\
  \textbf{Monotonic XGBoost} — 23 positive \& 8 negative domain-knowledge constraints\\
  \textbf{Walk-Forward CV} — $K=5$ expanding temporal folds (2016--2026), LGB retrained per fold\\
  \textbf{Quantile Regression} — LightGBM pinball loss at 7 quantiles ($\tau = 0.05,\ldots,0.95$)
};

% 5. Statistical evaluation
\node[eval, below=of ext] (eval) {%
  \textbf{Statistical Evaluation Framework}\\[2pt]
  \textbf{DM Test} + Newey-West HAC ($h=8$) \textbullet{}
  \textbf{Block Bootstrap} CI ($n=5000$ replications, 95\%) \textbullet{}
  \textbf{MCS} ($\alpha=0.10$)\\
  \textbf{Bonferroni correction} ($k=6$ tests) \textbullet{}
  \textbf{Cohen's $d$} effect sizes \textbullet{}
  \textbf{Ablation} (7 feature groups, 20 Optuna trials each)\\
  \textbf{Conformal Prediction} — MAPIE \texttt{SplitConformalRegressor}, 90\% nominal coverage
};

%% ── Side branches (explainability + rain) ────────────────

};

%% ── Economic analysis ─────────────────────────────────────

\node[econ, below=3.2cm of eval] (econ) {%

  \textbf{Gini Coefficient} — equity of prediction error across 10 ICC member nations (4 tiers)\\

  \textbf{ICC Revenue Context} — BCCI 38.5\% of ICC earnings vs.\ Associate nation budgets
};

%% ── Output / Conclusions ─────────────────────────────────

  Stacking (best): RMSE~43.39 \textbullet{} Walk-forward $\Delta$RMSE~$-20.44\pm2.79$ \textbullet{} EV at WC semi-final: \$1.27M/match\\
  All ML-vs-DLS DM tests: $p<0.001$ \textbullet{} Bootstrap 95\% CI: $[-23.94,\,-18.72]$ runs
};
